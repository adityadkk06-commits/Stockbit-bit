"""
ACCUMULATION SCREENER — Early accumulation phase stock detector
================================================================

Goal: find stocks where volume has already picked up but price hasn't
run yet. This flags smart money quietly accumulating before a breakout —
the opposite of an ARA hunter, which catches stocks that have ALREADY
flown (and the later you enter, the bigger the risk).

USAGE (integration with your existing bot):
---------------------------------------------
This module does NOT fetch data itself — it just takes data you already
have (from whatever data source your bot currently uses: RTI scrape,
Stockbit API, yfinance, etc.) and returns an accumulation score.

    from accumulation_screener import score_accumulation, AccumulationInput

    inp = AccumulationInput(
        ticker="XXXX",
        close=df["close"],          # pandas Series, chronological order (old -> new)
        volume=df["volume"],        # pandas Series
        value=df["value"],          # pandas Series (transaction value in Rupiah)
        freq=df["freq"],            # pandas Series (transaction frequency)
        foreign_net=df["foreign_net"],  # pandas Series (daily foreign net buy/sell)
    )
    result = score_accumulation(inp)

    if result.is_candidate:
        print(f"{inp.ticker}: score {result.score}/100 - {result.tags}")

Each Series needs at least ~25 days of recent data (to compute MA20, BB
width, etc). The most recent data point must be at the LAST index/position
of the series.

TELEGRAM BOT INTEGRATION:
---------------------------
Call score_accumulation() for each ticker inside your existing daily scan
loop (the one currently used for the ARA hunter), then filter on
result.is_candidate, sort by result.score descending, and format into a
Telegram message using the template at the bottom of this file
(format_telegram_message).
"""

from dataclasses import dataclass, field
import pandas as pd
import numpy as np


# ============================================================
# THRESHOLD CONFIG — tune these based on your own backtest results
# ============================================================

CONFIG = {
    "price_change_min": -2.0,      # min 1D price change %
    "price_change_max": 4.0,       # max 1D price change % (still "quiet")
    "volume_ratio_min": 2.0,       # today's volume vs 20D average
    "value_min_rupiah": 3_000_000_000,   # minimum transaction value (3 billion)
    "bb_period": 20,                # Bollinger Band period
    "bb_width_percentile_max": 30,  # BB width must be below this percentile (narrow)
    "foreign_streak_days": 3,       # consecutive days of positive foreign net buy
    "ma_period": 20,
    "ma_fresh_cross_lookback": 5,   # cross must have happened within the last N days
}


@dataclass
class AccumulationInput:
    ticker: str
    close: pd.Series
    volume: pd.Series
    value: pd.Series
    freq: pd.Series
    foreign_net: pd.Series  # daily foreign net buy(+)/sell(-), in Rupiah


@dataclass
class AccumulationResult:
    ticker: str
    score: int
    is_candidate: bool
    tags: list = field(default_factory=list)
    details: dict = field(default_factory=dict)


def _pct_change_today(series: pd.Series) -> float:
    if len(series) < 2:
        return 0.0
    prev, curr = series.iloc[-2], series.iloc[-1]
    if prev == 0:
        return 0.0
    return (curr - prev) / prev * 100


def _volume_ratio(volume: pd.Series, period: int = 20) -> float:
    if len(volume) < period + 1:
        return 0.0
    avg = volume.iloc[-(period + 1):-1].mean()
    if avg == 0:
        return 0.0
    return volume.iloc[-1] / avg


def _bollinger_width(close: pd.Series, period: int = 20) -> float:
    if len(close) < period:
        return np.nan
    window = close.iloc[-period:]
    ma = window.mean()
    std = window.std()
    upper, lower = ma + 2 * std, ma - 2 * std
    if ma == 0:
        return np.nan
    return (upper - lower) / ma * 100  # width as % of MA


def _bb_width_history(close: pd.Series, period: int = 20) -> pd.Series:
    """Compute rolling BB width so it can be compared against its own percentile."""
    ma = close.rolling(period).mean()
    std = close.rolling(period).std()
    upper, lower = ma + 2 * std, ma - 2 * std
    width = (upper - lower) / ma * 100
    return width.dropna()


def _ma_fresh_cross(close: pd.Series, period: int = 20, lookback: int = 5) -> bool:
    """True if price just crossed above the MA within the last N days
    (not a stock that's been above MA for a long time / an aging trend)."""
    if len(close) < period + lookback + 1:
        return False
    ma = close.rolling(period).mean()
    above = close > ma
    recent = above.iloc[-(lookback + 1):]
    # fresh cross = there's a False -> True transition in this window, and it's True now
    return recent.iloc[-1] and (not recent.iloc[0]) and recent.any()


def _foreign_streak(foreign_net: pd.Series, days: int = 3) -> bool:
    if len(foreign_net) < days:
        return False
    recent = foreign_net.iloc[-days:]
    return bool((recent > 0).all())


def _freq_rising_small_lot(freq: pd.Series, volume: pd.Series, period: int = 10) -> bool:
    """Frequency rising but avg lot/transaction small = gradual accumulation
    (many small orders), as opposed to distribution/pump (large orders at once)."""
    if len(freq) < period + 1 or len(volume) < period + 1:
        return False
    freq_now = freq.iloc[-1]
    freq_avg = freq.iloc[-(period + 1):-1].mean()
    if freq_avg == 0:
        return False
    freq_rising = freq_now > freq_avg * 1.3

    lot_now = volume.iloc[-1] / freq_now if freq_now else 0
    lot_avg = (volume.iloc[-(period + 1):-1].sum() / freq.iloc[-(period + 1):-1].sum()) \
        if freq.iloc[-(period + 1):-1].sum() else 0
    lot_small = lot_now <= lot_avg * 1.2 if lot_avg else False

    return freq_rising and lot_small


def score_accumulation(inp: AccumulationInput, config: dict = None) -> AccumulationResult:
    cfg = {**CONFIG, **(config or {})}
    tags = []
    score = 0
    details = {}

    # 1. Price change still quiet (weight 20)
    pct = _pct_change_today(inp.close)
    details["price_change_pct"] = round(pct, 2)
    if cfg["price_change_min"] <= pct <= cfg["price_change_max"]:
        score += 20
        tags.append("quiet_price")

    # 2. High volume ratio (weight 25 — most important signal)
    vol_ratio = _volume_ratio(inp.volume)
    details["volume_ratio"] = round(vol_ratio, 2)
    if vol_ratio >= cfg["volume_ratio_min"]:
        score += 25
        tags.append(f"volume_{vol_ratio:.1f}x")

    # 3. Transaction value liquid enough (weight 10)
    val_today = inp.value.iloc[-1] if len(inp.value) else 0
    details["value_today"] = val_today
    if val_today >= cfg["value_min_rupiah"]:
        score += 10
        tags.append("liquid")

    # 4. Fresh MA20 cross (weight 15)
    fresh_cross = _ma_fresh_cross(inp.close, cfg["ma_period"], cfg["ma_fresh_cross_lookback"])
    details["fresh_ma_cross"] = fresh_cross
    if fresh_cross:
        score += 15
        tags.append("fresh_cross_MA20")

    # 5. BB width narrow vs history (weight 15)
    bb_hist = _bb_width_history(inp.close, cfg["bb_period"])
    bb_narrow = False
    if len(bb_hist) >= 20:
        current_width = bb_hist.iloc[-1]
        percentile = (bb_hist < current_width).mean() * 100
        details["bb_width_percentile"] = round(percentile, 1)
        bb_narrow = percentile <= cfg["bb_width_percentile_max"]
    if bb_narrow:
        score += 15
        tags.append("tight_consolidation")

    # 6. Foreign net buy streak (weight 10)
    foreign_ok = _foreign_streak(inp.foreign_net, cfg["foreign_streak_days"])
    details["foreign_streak"] = foreign_ok
    if foreign_ok:
        score += 10
        tags.append("foreign_accum")

    # 7. Freq rising + small avg lot (weight 5)
    freq_pattern = _freq_rising_small_lot(inp.freq, inp.volume)
    details["freq_pattern_ok"] = freq_pattern
    if freq_pattern:
        score += 5
        tags.append("rising_freq_small_lot")

    # Candidate if score >= 60 AND volume ratio + quiet price are BOTH true
    # (these two conditions are non-negotiable, the rest just add to the score)
    is_candidate = (
        score >= 60
        and vol_ratio >= cfg["volume_ratio_min"]
        and cfg["price_change_min"] <= pct <= cfg["price_change_max"]
    )

    return AccumulationResult(
        ticker=inp.ticker,
        score=score,
        is_candidate=is_candidate,
        tags=tags,
        details=details,
    )


def format_telegram_message(results: list) -> str:
    """Format a list of AccumulationResult into a Telegram message, styled
    similarly to your ARA Hunter output."""
    candidates = sorted(
        [r for r in results if r.is_candidate], key=lambda r: -r.score
    )
    if not candidates:
        return "🔍 ACCUMULATION SCANNER\n\nNo accumulation candidates today."

    lines = [
        "🔍 ACCUMULATION SCANNER",
        f"✅ {len(candidates)} stocks in early accumulation phase detected\n",
    ]
    for i, r in enumerate(candidates, 1):
        d = r.details
        lines.append(
            f"{i}. {r.ticker}\n"
            f"Score: {r.score}/100\n"
            f"Chg: {d.get('price_change_pct')}%  VolRatio: {d.get('volume_ratio')}x\n"
            f"Tags: {', '.join(r.tags)}\n"
        )
    return "\n".join(lines)
